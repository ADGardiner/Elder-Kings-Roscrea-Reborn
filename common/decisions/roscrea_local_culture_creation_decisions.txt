# High Rock
create_local_culture_roscrea_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_realm.dds"

	desc = create_local_culture_roscrea_decision_desc
	major = yes

	ai_check_interval = 120

	is_shown = {
		is_landed = yes
		is_independent_ruler = yes
		exists = capital_province

		create_local_culture_decision_is_shown_scripted_trigger = { REGION = roscrea }
	}

	is_valid = {
		create_local_culture_decision_is_valid_scripted_trigger = { REGION = roscrea }
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	effect = {
		create_local_culture_decision_scripted_effects = { REGION = roscrea }

		if = {
			limit = { capital_province = { geographical_region = custom_nativity_roscrea } }
			custom_tooltip = create_local_culture_roscrea_decision_courtiers_embrace_new_culture
			custom_tooltip = create_local_culture_roscrea_decision_capital_embrace_new_culture
		}
		custom_tooltip = create_local_culture_roscrea_decision_provinces_embrace_new_culture
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
